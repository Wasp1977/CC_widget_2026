    69→    69→// ---- Helpers ----
    70→    70→function getAwtColor(value: number, sla: number): string {
    71→    71→  if (sla === 0) return 'text-zinc-400';
    72→    72→  if (value === 0) return 'text-emerald-600 dark:text-emerald-400';
    73→    73→  if (value <= sla * 0.5) return 'text-emerald-600 dark:text-emerald-400';
    74→    74→  if (value <= sla) return 'text-amber-600 dark:text-amber-400';
    75→    75→  return 'text-red-600 dark:text-red-400';
    76→    76→}
    77→    77→
    78→    78→function getAwtBg(value: number, sla: number): string {
    79→    79→  if (sla === 0) return '';
    80→    80→  if (value === 0) return 'bg-emerald-50 dark:bg-emerald-950/30';
    81→    81→  if (value <= sla * 0.5) return 'bg-emerald-50 dark:bg-emerald-950/30';
    82→    82→  if (value <= sla) return 'bg-amber-50 dark:bg-amber-950/30';
    83→    83→  return 'bg-red-50 dark:bg-red-950/30';
    84→    84→}

   104→   104→function fmt(s: number): string {
   105→   105→  if (s === 0) return '0:00';
   106→   106→  const m = Math.floor(s / 60);
   107→   107→  const sec = s % 60;
   108→   108→  return `${m}:${sec.toString().padStart(2, '0')}`;
   109→   109→}

   129→   129→function useRealtimeData() {
   130→   130→  const [queues, setQueues] = useState<Queue[]>(INITIAL_QUEUES);
   131→   131→  const [agents, setAgents] = useState<Agent[]>(INITIAL_QUEUES.length > 0 ? INITIAL_AGENTS : []);
   132→   132→  const tickRef = useRef<ReturnType<typeof setInterval>>();
   133→   133→
   134→   134→  const changeAgentStatus = useCallback((agentId: string, newStatus: Agent['status']) => {
   135→   135→    setAgents(prev => prev.map(a =>
   136→   136→      a.id === agentId ? { ...a, status: newStatus, statusSince: new Date().toISOString() } : a
   137→   137→    ));
   138→   138→  }, []);
   139→   139→
   140→   140→  useEffect(() => {
   141→   141→    tickRef.current = setInterval(() => {
   142→   142→      setQueues(prev => prev.map(q => {
   143→   143→        if (q.slaSeconds === 0) return q;
   144→   144→        return {
   145→   145→          ...q,
   146→   146→          awtCurrent: Math.max(0, q.awtCurrent + Math.round((Math.random() - 0.5) * 16)),
   147→   147→          awt10min: Math.max(0, q.awt10min + Math.round((Math.random() - 0.5) * 10)),
   148→   148→          awtDay: Math.max(0, q.awtDay + Math.round((Math.random() - 0.5) * 4)),
   149→   149→          queueDepth: Math.max(0, q.queueDepth + Math.round((Math.random() - 0.5) * 3)),
   150→   150→          availAgents: Math.max(0, Math.min(q.totalAgents, q.availAgents + (Math.random() < 0.05 ? (Math.random() < 0.5 ? 1 : -1) : 0))),
   151→   151→        };
   152→   152→      }));
   153→   153→
   154→   154→      setAgents(prev => prev.map(a => {
   155→   155→        if (a.status === 'offline') return a;
   156→   156→        let newStatus = a.status;
   157→   157→        const r = Math.random();
   158→   158→        if (a.status === 'online' && r < 0.05) newStatus = 'break';
   159→   159→        else if (a.status === 'break' && r < 0.08) newStatus = 'online';
   160→   160→        if (newStatus !== a.status) return { ...a, status: newStatus, statusSince: new Date().toISOString() };
   161→   161→        return a;
   162→   162→      }));
   163→   163→    }, 30000);
   164→   164→
   165→   165→    return () => { if (tickRef.current) clearInterval(tickRef.current); };
   166→   166→  }, []);
   167→   167→
   168→   168→  return { queues, agents, changeAgentStatus };
   169→   169→}
